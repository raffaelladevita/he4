      function cross(eg,t,xm2p,tgj,epsf)
      implicit double precision (a-h,o-z)
      double complex Y

      common /mbin/ xmmin(200),xmmax(200),xmid(200)
c      dimension wynr(300,300),errr(300,300)
      common /wave/ wynr(300,300),errr(300,300)
      common /bin/ ie,it,mass,lmax,itot
      common /ener/ emin,emax




      pi = 4.d0*datan(1.d0)


      xmn = 0.93827
      xmp = 0.13857


      if (xm2p.lt.xmmin(1)) then
      imbin =1
      endif
      do i=1,mass
      if ((xm2p.gt.xmmin(i)).and.(xm2p.le.xmmax(i))) then
      imbin = i
      endif
      enddo
      if (xm2p.gt.xmmax(mass)) then
      imbin = mass
      endif

      sum13 = 0.d0
      ia = 0
      do l=0,lmax
      do m=0,l
      ia = ia + 1
      if (m.eq.0) then
      eps = 1.
      else
      eps = 1./2.
      endif
      sum13 = sum13 +
     $    dsqrt(4.*pi)
     $ *dble((       Y(l, m,tgj,epsf)
     $     + dconjg(Y(l, m,tgj,epsf)) )/2. )
     $  *(wynr(imbin,ia)/eps)
      enddo
      enddo


c----------------

      cross = sum13

      return
      end
